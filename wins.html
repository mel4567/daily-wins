<!DOCTYPE html>

<html lang="en">

<head>

<meta charset="UTF-8">

<meta name="viewport" content="width=device-width, initial-scale=1.0">

<title>Your Daily Wins</title>

<script crossorigin src="https://cdnjs.cloudflare.com/ajax/libs/react/18.2.0/umd/react.production.min.js"></script>

<script crossorigin src="https://cdnjs.cloudflare.com/ajax/libs/react-dom/18.2.0/umd/react-dom.production.min.js"></script>

<script src="https://cdnjs.cloudflare.com/ajax/libs/babel-standalone/7.23.5/babel.min.js"></script>

<script src="https://cdn.tailwindcss.com"></script>

</head>

<body>

<div id="root"></div>

<script type="text/babel">

const { useState, useEffect } = React;

// Lucide icons as inline SVGs

const Check = ({ className }) => (

<svg className={className} width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">

<polyline points="20 6 9 17 4 12"></polyline>

</svg>

);

const Sparkles = ({ className }) => (

<svg className={className} width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">

<path d="m12 3-1.912 5.813a2 2 0 0 1-1.275 1.275L3 12l5.813 1.912a2 2 0 0 1 1.275 1.275L12 21l1.912-5.813a2 2 0 0 1 1.275-1.275L21 12l-5.813-1.912a2 2 0 0 1-1.275-1.275L12 3Z"></path>

<path d="M5 3v4"></path>

<path d="M19 17v4"></path>

<path d="M3 5h4"></path>

<path d="M17 19h4"></path>

</svg>

);

const Trophy = ({ className }) => (

<svg className={className} width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">

<path d="M6 9H4.5a2.5 2.5 0 0 1 0-5H6"></path>

<path d="M18 9h1.5a2.5 2.5 0 0 0 0-5H18"></path>

<path d="M4 22h16"></path>

<path d="M10 14.66V17c0 .55-.47.98-.97 1.21C7.85 18.75 7 20.24 7 22"></path>

<path d="M14 14.66V17c0 .55.47.98.97 1.21C16.15 18.75 17 20.24 17 22"></path>

<path d="M18 2H6v7a6 6 0 0 0 12 0V2Z"></path>

</svg>

);

const Calendar = ({ className }) => (

<svg className={className} width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">

<rect x="3" y="4" width="18" height="18" rx="2" ry="2"></rect>

<line x1="16" y1="2" x2="16" y2="6"></line>

<line x1="8" y1="2" x2="8" y2="6"></line>

<line x1="3" y1="10" x2="21" y2="10"></line>

</svg>

);


const DailyChallengeTracker = () => {

const creativeIdeas = [

'Doodle for 2 minutes',

'Take a photo of something beautiful',

'Write 3 sentences about anything',

'Rearrange something on your desk',

'Color in a coloring book',

'Make up a silly song',

'Dance to one song',

'Try a new recipe twist',

'Snap a selfie with a fun filter',

'Write down a dream you remember'

];


const selfCareIdeas = [

'Take 3 deep breaths',

'Stretch for 2 minutes',

'Listen to your favorite song',

'Make yourself a nice drink',

'Put on lotion or lip balm',

'Look out the window for a minute',

'Text someone you love',

'Tidy one small space',

'Light a candle or open a window',

'Give yourself a compliment'

];


const initialChallenges = [

{ id: 1, category: 'Wellness', text: 'Drink 3 glasses of water', completed: false, emoji: '💧' },

{ id: 2, category: 'Wellness', text: 'Take your meds', completed: false, emoji: '💊' },

{ id: 3, category: 'Movement', text: 'Move your body for 10 min', completed: false, emoji: '🚶‍♀️' },

{ id: 4, category: 'Creative', text: 'Choose a creative challenge', completed: false, emoji: '🎨', type: 'creative' },

{ id: 5, category: 'Self-Care', text: 'Choose a self-care moment', completed: false, emoji: '✨', type: 'selfcare' },

{ id: 6, category: 'Mindful', text: '5 min breathing or meditation', completed: false, emoji: '🧘‍♀️' },

];


const [challenges, setChallenges] = useState(() => {

const saved = localStorage.getItem('dailyChallenges');

return saved ? JSON.parse(saved) : initialChallenges;

});


const [streak, setStreak] = useState(() => {

const saved = localStorage.getItem('streak');

return saved ? parseInt(saved) : 0;

});


const [points, setPoints] = useState(() => {

const saved = localStorage.getItem('points');

return saved ? parseInt(saved) : 0;

});


const [lastResetDate, setLastResetDate] = useState(() => {

return localStorage.getItem('lastResetDate') || new Date().toDateString();

});


const [showConfetti, setShowConfetti] = useState(false);

const [expandedIdea, setExpandedIdea] = useState(null);

const [selectedCreative, setSelectedCreative] = useState(() => {

return localStorage.getItem('selectedCreative') || null;

});

const [selectedSelfCare, setSelectedSelfCare] = useState(() => {

return localStorage.getItem('selectedSelfCare') || null;

});


useEffect(() => {

const today = new Date().toDateString();

if (lastResetDate !== today) {

const wasAllCompleted = challenges.every(c => c.completed);

if (wasAllCompleted) {

setStreak(prev => prev + 1);

}

setChallenges(initialChallenges);

setSelectedCreative(null);

setSelectedSelfCare(null);

setLastResetDate(today);

localStorage.setItem('lastResetDate', today);

localStorage.removeItem('selectedCreative');

localStorage.removeItem('selectedSelfCare');

}

}, []);


useEffect(() => {

localStorage.setItem('dailyChallenges', JSON.stringify(challenges));

}, [challenges]);


useEffect(() => {

localStorage.setItem('streak', streak.toString());

}, [streak]);


useEffect(() => {

localStorage.setItem('points', points.toString());

}, [points]);


useEffect(() => {

if (selectedCreative) {

localStorage.setItem('selectedCreative', selectedCreative);

}

}, [selectedCreative]);


useEffect(() => {

if (selectedSelfCare) {

localStorage.setItem('selectedSelfCare', selectedSelfCare);

}

}, [selectedSelfCare]);


const completedCount = challenges.filter(c => c.completed).length;

const totalCount = challenges.length;

const progress = (completedCount / totalCount) * 100;


const selectChallenge = (challengeId, type, selectedText) => {

if (type === 'creative') {

setSelectedCreative(selectedText);

} else if (type === 'selfcare') {

setSelectedSelfCare(selectedText);

}

setChallenges(challenges.map(c =>

c.id === challengeId ? { ...c, text: selectedText } : c

));

setExpandedIdea(null);

};


const toggleChallenge = (id) => {

const challenge = challenges.find(c => c.id === id);

const wasCompleted = challenge.completed;

setChallenges(challenges.map(c =>

c.id === id ? { ...c, completed: !c.completed } : c

));

if (!wasCompleted) {

setPoints(prev => prev + 1);

setShowConfetti(true);

setTimeout(() => setShowConfetti(false), 1000);

} else {

setPoints(prev => Math.max(0, prev - 1));

}

if (expandedIdea === id) {

setExpandedIdea(null);

}

};


return (

<div className="min-h-screen bg-gradient-to-br from-purple-50 via-pink-50 to-blue-50 p-4 sm:p-8">

<div className="max-w-2xl mx-auto">

<div className="text-center mb-8">

<h1 className="text-4xl font-bold text-purple-900 mb-2 flex items-center justify-center gap-2">

<Sparkles className="text-pink-500" />

Your Daily Wins

<Sparkles className="text-pink-500" />

</h1>

<p className="text-gray-600">Small steps, big impact 💜</p>

</div>


<div className="bg-white rounded-2xl shadow-lg p-6 mb-6">

<div className="flex items-center justify-between mb-3">

<span className="text-lg font-semibold text-gray-700">Today's Progress</span>

<div className="flex items-center gap-4">

<div className="text-center">

<div className="text-xs text-gray-500">Points</div>

<div className="text-2xl font-bold text-pink-600">⭐ {points}</div>

</div>

<span className="text-2xl font-bold text-purple-600">{completedCount}/{totalCount}</span>

</div>

</div>

<div className="w-full bg-gray-200 rounded-full h-4 overflow-hidden">

<div

className="bg-gradient-to-r from-purple-500 to-pink-500 h-4 rounded-full transition-all duration-500 ease-out"

style={{ width: `${progress}%` }}

/>

</div>

{completedCount === totalCount && totalCount > 0 && (

<div className="mt-4 text-center">

<Trophy className="inline-block text-yellow-500 w-8 h-8 animate-bounce" />

<p className="text-lg font-bold text-purple-600 mt-2">Amazing! All done! 🎉</p>

</div>

)}

</div>


<div className="space-y-3">

{challenges.map((challenge) => (

<div key={challenge.id}>

<button

onClick={() => toggleChallenge(challenge.id)}

className={`w-full text-left p-5 rounded-xl transition-all duration-300 transform hover:scale-102 ${

challenge.completed

? 'bg-gradient-to-r from-green-400 to-emerald-400 shadow-md'

: 'bg-white hover:shadow-lg'

}`}

>

<div className="flex items-center gap-4">

<div className={`flex-shrink-0 w-12 h-12 rounded-full flex items-center justify-center text-2xl ${

challenge.completed ? 'bg-white' : 'bg-purple-100'

}`}>

{challenge.completed ? <Check className="text-green-600 w-6 h-6" /> : challenge.emoji}

</div>

<div className="flex-1">

<div className="text-xs font-semibold text-purple-600 mb-1">{challenge.category}</div>

<div className={`text-lg font-medium ${

challenge.completed ? 'text-white line-through' : 'text-gray-800'

}`}>

{challenge.text}

</div>

{challenge.type && !challenge.completed && (

<button

onClick={(e) => {

e.stopPropagation();

setExpandedIdea(challenge.id);

}}

className="text-xs text-purple-500 hover:text-purple-700 mt-1 underline"

>

{challenge.type === 'creative' && !selectedCreative ? 'Pick your challenge!' :

challenge.type === 'selfcare' && !selectedSelfCare ? 'Pick your challenge!' :

'Change challenge'}

</button>

)}

</div>

</div>

</button>

{challenge.type && expandedIdea === challenge.id && !challenge.completed && (

<div className="mt-2 bg-purple-50 rounded-xl p-4">

<p className="text-sm font-semibold text-purple-700 mb-2">Pick one to do today:</p>

<div className="grid grid-cols-1 gap-2">

{(challenge.type === 'creative' ? creativeIdeas : selfCareIdeas).map((idea, idx) => (

<button

key={idx}

onClick={() => selectChallenge(challenge.id, challenge.type, idea)}

className="text-sm text-left text-gray-700 bg-white hover:bg-purple-100 rounded-lg px-3 py-2 transition-colors"

>

• {idea}

</button>

))}

</div>

<button

onClick={() => setExpandedIdea(null)}

className="text-xs text-purple-500 hover:text-purple-700 mt-3 underline"

>

Cancel

</button>

</div>

)}

</div>

))}

</div>


<div className="mt-8 text-center bg-white rounded-2xl shadow-lg p-6">

<Calendar className="inline-block text-purple-500 w-6 h-6 mb-2" />

<p className="text-gray-600">Perfect Days Streak</p>

<p className="text-4xl font-bold text-purple-600 mt-2">{streak} 🔥</p>

</div>


{showConfetti && (

<div className="fixed inset-0 pointer-events-none flex items-center justify-center">

<div className="text-6xl animate-ping">⭐</div>

</div>

)}

</div>

</div>

);

};


ReactDOM.render(<DailyChallengeTracker />, document.getElementById('root'));

</script>

</body>

</html>
